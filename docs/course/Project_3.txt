Training on Lab Workstations
Conda:
/share/instsww/anaconda3/bin/ conda init
Restart terminal
conda config --set auto_activate_base  false
conda deactivate
conda create -n myenv python=3.14 -c conda-forge
conda activate myenv
conda install -c nvidia/label/cuda -13.2.0 "cuda-toolkit=13.2" " cuda-nvcc=13.2" "cuda-version=13.2"
conda install cmake
Train and bundle model
git clone https:// github.com/jonas-eschmann/viobench
pip install -e viobench[train]
cd viobench/viobench/solution/examples/python/ tiny_cnn
python train.py --generate 8 --epochs 10
python3 -m viobench bundle solution -o tiny_cnn.wasm
python3 -m viobench check tiny_cnn.wasm
